$ SHELL-POP
#+begin_src emacs-lisp
(use-package shell-pop :ensure t
    :bind ("M-`" . shell-pop)
    :config
    (with-eval-after-load 'term
    (define-key term-raw-map (kbd "M-`") #'shell-pop))  ;; needed for my personal keybind
    (setq   shell-pop-universal-key "M-`")              ;; the keybind to toggle the shell window
    (setq   shell-pop-window-position "bottom")
    (setq   shell-pop-full-span nil)
    (setq   shell-pop-term-shell shell-file-name)
    (setq   shell-pop-window-size 50)
    (setopt shell-pop-shell-type '("ansi-term" "*ansi-term*" (lambda ()  ;; `setopt is important
                                   (ansi-term shell-pop-term-shell))))
    (setq   shell-pop-autocd-to-working-dir nil))
#+end_src
