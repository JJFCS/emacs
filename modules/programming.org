! PROGRAMMING
$ TABS & INDENTS & SPACES & WHITESPACES & ETC
#+begin_src emacs-lisp
(setq whitespace-style
    '(face indentation newline tabs tab-mark trailing
        spaces
        space-before-tab
        space-after-tab
        space-mark
    )
)

(electric-indent-mode 1)

(setq python-indent-guess-indent-offset nil)

(setq-default tab-always-indent 'complete) (setq-default indent-tabs-mode nil)
(setq-default tab-width 4)
(setq-default standard-indent 4)
#+end_src

$ GIT & VC & MAGIT & ETC
#+begin_src emacs-lisp
(use-package git-gutter :ensure t)
(use-package magit      :ensure t :defer t)
#+end_src

$ VARIOUS PROGRAMMING LANGUAGES
#+begin_src emacs-lisp
(use-package solidity-mode :ensure t)
(use-package ess           :ensure t)
(use-package ess-r-mode
        :ensure ess
        :mode ("\\.[rR]\\'" . ess-r-mode)
        :commands R
        :init
        (setq ess-style 'RStudio)
        (setq ess-indent-offset 4)
        (setq ess-ask-for-ess-directory nil)
        (setq ess-eval-visibly 'nowait)
        (setq inferior-R-args "--no-save --no-restore-data")
        :hook (
            (ess-r-mode . corfu-mode)
            (inferior-ess-r-mode . corfu-mode)
            (ess-r-mode . (lambda () (setq-local ess-indent-offset 4)))
        )
)
#+end_src

! HOOKS
#+begin_src emacs-lisp
(add-hook 'prog-mode-hook
    (lambda ()
        (git-gutter-mode 1) (corfu-mode 1)
        (whitespace-mode 1) (indent-bars-mode 1)
        (setq-local tab-width 4)
        (setq-local indent-tabs-mode t)
    )
)
#+end_src
