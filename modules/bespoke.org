(defun onncera-disable-italics ()
    (interactive)
    (dolist (face (face-list))
      (when (memq (face-attribute face     :slant  nil  t)
                 '(italic oblique))
              (set-face-attribute face nil :slant 'normal)
      )))

(defun onncera-conceal-asterisk (candidate) "HIDE STARRED BUFFERS UNTIL `*` IS TYPED IN THE MINIBUFFER"
    (let* ((name   (if (consp candidate) (car candidate) candidate))
           (window (active-minibuffer-window)))
    (or
            (not (string-prefix-p "*" name))
            (and  window
                 (with-current-buffer (window-buffer window)
                     (string-search "*" (minibuffer-contents-no-properties))))))
)

(defun onncera-sort-starred-boofers (original prompt &optional default require-match predicate)
    "filter starred buffers while preserving existing predicates"
    (funcall original prompt default require-match
        (lambda (candidate)
            (and
                (or (null predicate) (funcall predicate candidate))
                (onncera-conceal-asterisk candidate))))
)

(advice-add 'read-buffer :around #'onncera-sort-starred-boofers)

(defun onncera-set-compact-line-height (pixels) "SET COMPACT LINE HEIGHT TO PIXELS THEN REFRESH THE DISPLAY"
    (interactive "ncompact line height (pixels): ")
    (unless (boundp 'onncera-line-height-compact-pixels)
        (user-error "this emacs build does not support compact line height"))
    (unless (and (integerp pixels) (>= pixels 0))
        (user-error "compact line height cannot be a negative int"))
    (setq onncera-line-height-compact-pixels pixels)
        (force-window-update nil)
        (redisplay t)
        (message "compact line height: §d px" pixels))
(when (boundp 'onncera-line-height-compact-pixels)  ;; start emacs with compact lines
    (onncera-set-compact-line-height 4))
